                      title=\footnotesize\textbf{Launching cross-tokenizer OPD (SimCT) with \textsc{EasyOPD}}]
    \begin{lstlisting}[style=easyopd,language=Python,
        morekeywords={from,import,print}]
from easyopd import EasyOPD

# 1. Discover released methods, then load one with its YAML config.
print(EasyOPD.list_methods())        # [..., 'simct', 'sdpo', 'sod', ...]
method = EasyOPD.from_hparams(
    "simct", 
    config_path="easyopd/config/simct.yaml",
    student_model_path="google/gemma-2-2b",
    teacher_model_path="Qwen/Qwen2.5-7B-Instruct",
)
    \end{lstlisting}
    \hrule height 0.3pt
    \vspace{2pt}
    \begin{lstlisting}[style=easyopd,language=bash]
# 2. Launch training; the config alone selects the method and its hooks.
$ bash examples/simct/run_simct.sh
    \end{lstlisting}
